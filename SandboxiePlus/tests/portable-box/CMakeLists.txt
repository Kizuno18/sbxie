cmake_minimum_required(VERSION 3.16)
project(PortableBoxTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
enable_testing()
add_executable(portable_box_test portable_box_test.cpp)
target_link_libraries(portable_box_test PRIVATE Qt6::Core)
add_test(NAME portable_box COMMAND portable_box_test)
