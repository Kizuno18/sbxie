cmake_minimum_required(VERSION 3.16)
project(SharedAccessTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
enable_testing()
set(widgetDir "${CMAKE_CURRENT_SOURCE_DIR}/../../SandMan/Windows")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${widgetDir}/SharedAccessWidget.cpp")
file(READ "${widgetDir}/SharedAccessWidget.cpp" widgetSource)
string(REPLACE "#include \"stdafx.h\"" "#include <QtWidgets>" widgetSource "${widgetSource}")
string(REPLACE "#include \"../QSbieAPI/Sandboxie/SbieIni.h\"" "#include \"fake_sbie_ini.h\"" widgetSource "${widgetSource}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/SharedAccessWidget.cpp" "${widgetSource}")
add_executable(shared_access_test shared_access_test.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/SharedAccessWidget.cpp" "${widgetDir}/SharedAccessWidget.h")
target_include_directories(shared_access_test PRIVATE "${widgetDir}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(shared_access_test PRIVATE Qt6::Widgets)
add_test(NAME shared_access COMMAND shared_access_test)
set_tests_properties(shared_access PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
